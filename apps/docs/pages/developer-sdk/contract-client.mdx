# TypeScript SDK & Contract Client

## Installation

```bash
pnpm add @pulsepay/contract-client
```

## Initializing Client and Reading Stream State

```typescript
import { Client, networks } from "@pulsepay/contract-client";

const client = new Client({
  ...networks.testnet,
  contractId: "CBH3ATY3PU7MRK54BOLEHU2ANM67LXLALNR5TQP6MR44NFKOLLEFHRZW",
});

async function fetchWorkerBalance(streamId: bigint): Promise<bigint> {
  const stream = await client.get_stream({ stream_id: streamId });
  const now = BigInt(Math.floor(Date.now() / 1000));
  
  if (now >= stream.end_time) return stream.amount - stream.withdrawn_amount;
  
  const elapsed = now - stream.start_time;
  const totalDuration = stream.end_time - stream.start_time;
  const accrued = (stream.amount * elapsed) / totalDuration;
  
  return accrued - stream.withdrawn_amount;
}
```
